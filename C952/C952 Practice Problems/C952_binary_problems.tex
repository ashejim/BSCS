\question \textbf{Choose the best answer.}
        %%%%binary addition problem%%%%
        Given the following 8-bit integer binary variables:
            \begin{itemize}
            \item[] {\tt{X1}} = 0101 1111
            \item[] {\tt{X2}} = 0010 0011
            \end{itemize}        
        What is the value in X3 after the following command:
            \begin{itemize}
            \item[] {\tt{ADD X3, X2, X1}}
            \end{itemize}
		\begin{choices}
			\choice 0010 0000
			\choice Overflow error or carry flag
			\CorrectChoice 1000 0010
			\choice 0100 0001
		\end{choices}%
