\question \textbf{Choose the best answer.}
        How is the integer $-5_{10}$ represented in memory?
		\begin{choices}
			\choice 11110110
			\choice 11111101
			\choice 11111110
			\CorrectChoice 11111011
		\end{choices}%
		
		\begin{solution}
		    Assume that the \textit{two's complement} method is being used. Other  methods for \href{https://www.geeksforgeeks.org/representation-of-negative-binary-numbers/}{Representing negative binary numbers} exist, but they have ambiguous representations for zero. Hence \textit{two's complement} is used in computing.  
		    
		    \textbf{Two's complement method:}\\
		    To represent both positive and negative numbers, the most significant bit (MSB) is always the \textit{sign bit}. If the MSB is 0, there are no changes, e.g., $5=00000101$.
		    The MSB is 1 for negative numbers. If it can be calculated directly as follows:
		    The leftmost 1 represents $-2^{3}=-8$ (the four bits are 3, 2, 1, 0)
		    
            Now everything else is positive: 
            \begin{equation*}
                \begin{array}{rrrr}
                    3^{\text{rd}}\text{ bit}&= 0 \cdot 2^{2}& = 0\\
                    2^{\text{nd}} \text{ bit}&= 1 \cdot 2^{1}& = 2\\
                    1^{\text{st}} \text{ bit}&= 1 \cdot 2^{0}&= 1\\
                \end{array}
            \end{equation*}
            
            So 1011 $= (-8)+(0)+(2)+(1) = -5$\\
            
            Alternatively, you can find the one’s complement of $5 = 0000 0101$ (I need more than 4 bits for this):\\
            $1111 0101$ (just flip everything) and add $0000 0001$ to get the two’s complement:
            \begin{equation*}
                \begin{array}{B3}
                & \carry 1001 & 1111 \\
                -&0101 & 0101 \\ \hline
                &0100 & 1010 \\
                \end{array}
                \vspace{-7pt}
            \end{equation*}
            % 1111 1010
            % +        0001
            % --------------
            % 1111 1101
            I cover this problem \href{https://wgu.hosted.panopto.com/Panopto/Pages/Viewer.aspx?id=d658c97f-f918-4a62-89ee-ad2e0105d641}{here} around the 8:50 mark. (BTW, in this video, toward the end, I get confused around the 1-hour mark and say that 2.12 should be 5 -4 is the correct answer. It was late.)
            		    
		\end{solution}

\question \textbf{Choose the best answer.}
        An instruction set has 5 steps each taking 4 clock cycles to complete. 
        Approximately what is the average number of cycles needed to complete each instruction set using a pipelined process to complete 
        $2.15\times 10^{50}$ instruction sets?
        
		\begin{choices}
			\choice 20
			\choice 0
			\choice 1.25
			\CorrectChoice 4
			\choice ${4}\times 10^{50}$
		\end{choices}%
		
		\begin{solution}
		    For a pipelined process, we find the total number of cycles for $n$ instruction sets as follows:
    		    \begin{align}
    		        \text{total time} &= [\text{time for $1^{\text{st}}$ set}]+(n-1)(\text{time for slowest step})
    		    \end{align}
		    To find the average time, we simply divide this by the number of instruction sets, $n$:
    		    \begin{align}
    		        \text{average} &= \frac{[\text{time for $1^{\text{st}}$ set}]+(n-1)(\text{time for slowest step})}{n}
    		    \end{align}
		    Here, time = cycles and our slowest step takes $4$ cycles (they are all the same): 
    		    \begin{align}
    		        \text{average} &= \frac{[4\cdot 5]+(n-1)(4)}{n}\\
    		                       &= \frac{[4\cdot 5]+(2.15\cdot 10^{50}-1)(4)}{2.15\cdot 10^{50}}\\
    		                       &\approx 4
    		    \end{align}
    		For these types of problems, there's no need to plug in $n=2.15\times 10^{50}$. Note as $n$ gets large, the average tends toward the time of the slowest step.\\
            \begin{center}
            \hspace{20pt}\begin{tabular}{ c|c } 
                 $n$ & average \\ 
                 \hline
                 10 & 5.6 \\ 
                 100 & 4.16 \\ 
                 1000 & 4.016 \\ 
                 10000 & 4.0016 \\ 
                 \vdots & \vdots \\ 
                 $10^{\infty}$ & 4 \\ 
            \end{tabular}
            \end{center}
            Recall from calculus,
            \begin{align}
    		        \lim_{n\rightarrow\infty}\text{average}
    		        &=\lim_{n\rightarrow\infty} \frac{[4\cdot 5]+(n-1)(4)}{n}\\
    		        &=\lim_{n\rightarrow\infty}\cancelto{0}{\frac{[4\cdot 5]}{n}}
    		        +\lim_{n\rightarrow\infty}4\cancel{\frac{n}{n}}
    		        +\lim_{n\rightarrow\infty}\cancelto{0}{\frac{-4}{n}}\\
    		        &= 4
    		    \end{align}
    		
		\end{solution}
